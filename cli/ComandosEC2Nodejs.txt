//Clonar sin descargar el contenido
git clone --no-checkout https://github.com/JhoelCJ/TRAAJOSWEB.git

//Entrar a la carpeta del repositorio
cd TRAAJOSWEB

//Inicializar la descarga selectiva
git sparse-checkout init --cone

//Indicar la carpeta exacta
git sparse-checkout set wsCellphone

//Descargar los archivos
git checkout main
